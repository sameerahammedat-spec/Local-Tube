import { registerPlugin, Capacitor } from '@capacitor/core';

let androidDbPromise;

async function getAndroidDb() {
  if (!androidDbPromise) {
    androidDbPromise = (async () => {
      const sqlite = new SQLiteConnection(CapacitorSQLite);
      const consistency = await sqlite.checkConnectionsConsistency();
      const connection = await sqlite.isConnection('localtube', false);

      let db;

      if (consistency.result && connection.result) {
        db = await sqlite.retrieveConnection('localtube', false);
      } else {
        db = await sqlite.createConnection(
          'localtube', false, 'no-encryption', 1, false
        );
      }

      await db.open();

      // Safe migration: retain all existing videos and add source URI tracking.
      const columns = await db.query("PRAGMA table_info(videos)");
      const hasSourceUri = (columns.values || []).some(
        column => column.name === 'source_uri'
      );

      if (!hasSourceUri) {
        await db.execute("ALTER TABLE videos ADD COLUMN source_uri TEXT DEFAULT ''");
      }

      await db.execute(`
        CREATE UNIQUE INDEX IF NOT EXISTS videos_unique_source_uri
        ON videos(source_uri) WHERE source_uri <> '';
        CREATE TABLE IF NOT EXISTS videos (
          id INTEGER PRIMARY KEY AUTOINCREMENT,
          title TEXT NOT NULL,
          channel_title TEXT DEFAULT '',
          channel_id TEXT DEFAULT '',
          youtube_id TEXT DEFAULT '',
          youtube_url TEXT DEFAULT '',
          file_path TEXT DEFAULT '',
          thumbnail_url TEXT DEFAULT '',
          description TEXT DEFAULT '',
          tags TEXT DEFAULT '',
          created_at TEXT DEFAULT CURRENT_TIMESTAMP
        );

        CREATE UNIQUE INDEX IF NOT EXISTS videos_unique_local_path
        ON videos(file_path) WHERE file_path <> '';

        CREATE UNIQUE INDEX IF NOT EXISTS videos_unique_youtube_id
        ON videos(youtube_id) WHERE youtube_id <> '';

        CREATE TABLE IF NOT EXISTS bookmarks (
          id INTEGER PRIMARY KEY AUTOINCREMENT,
          video_id INTEGER NOT NULL UNIQUE,
          created_at TEXT DEFAULT CURRENT_TIMESTAMP,
          FOREIGN KEY(video_id) REFERENCES videos(id) ON DELETE CASCADE
        );

        CREATE TABLE IF NOT EXISTS settings (
          key TEXT PRIMARY KEY,
          value TEXT NOT NULL
        );

        CREATE TABLE IF NOT EXISTS video_notes (
          video_id INTEGER PRIMARY KEY,
          content TEXT NOT NULL DEFAULT '',
          updated_at TEXT DEFAULT CURRENT_TIMESTAMP,
          FOREIGN KEY(video_id) REFERENCES videos(id) ON DELETE CASCADE
        );

        CREATE TABLE IF NOT EXISTS watch_history (
          video_id INTEGER PRIMARY KEY,
          position_seconds REAL NOT NULL DEFAULT 0,
          duration_seconds REAL NOT NULL DEFAULT 0,
          completed INTEGER NOT NULL DEFAULT 0,
          last_watched_at TEXT DEFAULT CURRENT_TIMESTAMP,
          FOREIGN KEY(video_id) REFERENCES videos(id) ON DELETE CASCADE
        );

        CREATE INDEX IF NOT EXISTS watch_history_recent
        ON watch_history(last_watched_at DESC);
      `);

      return db;
    })().catch(error => {
      androidDbPromise = null;
      throw error;
    });
  }

  return androidDbPromise;
}

function mapVideo(row) {
  return {
    ...row,
    isLocal: Boolean(row.file_path),
    bookmarked: Boolean(row.bookmarked),
    fileExists: Boolean(row.file_path),
    progressSeconds: Number(row.position_seconds || 0),
    durationSeconds: Number(row.duration_seconds || 0),
    completed: Boolean(row.completed)
  };
}

async function androidListVideos(options = {}) {
  const db = await getAndroidDb();
  const {
    type = 'all',
    channelId = '',
    query = '',
    bookmarkedOnly = false
  } = options || {};

  let sql = `
    SELECT v.*,
      EXISTS (
        SELECT 1 FROM bookmarks b WHERE b.video_id = v.id
      ) AS bookmarked,
      COALESCE(h.position_seconds, 0) AS position_seconds,
      COALESCE(h.duration_seconds, 0) AS duration_seconds,
      COALESCE(h.completed, 0) AS completed
    FROM videos v
    LEFT JOIN watch_history h ON h.video_id = v.id
    WHERE 1 = 1
  `;

  const values = [];

  if (type === 'local') {
    sql += " AND v.file_path <> ''";
  } else if (type === 'youtube') {
    sql += " AND v.youtube_id <> '' AND v.file_path = ''";
  }

  if (channelId) {
    sql += ' AND v.channel_id = ?';
    values.push(String(channelId));
  }

  if (bookmarkedOnly) {
    sql += `
      AND EXISTS (
        SELECT 1 FROM bookmarks b WHERE b.video_id = v.id
      )
    `;
  }

  if (query && String(query).trim()) {
    sql += `
      AND (
        v.title LIKE ? OR
        v.channel_title LIKE ? OR
        v.tags LIKE ? OR
        v.description LIKE ?
      )
    `;
    const term = `%${String(query).trim()}%`;
    values.push(term, term, term, term);
  }

  sql += ' ORDER BY v.created_at DESC, v.id DESC LIMIT 1000';

  const result = await db.query(sql, values);
  return (result.values || []).map(mapVideo);
}

async function androidListChannels() {
  const db = await getAndroidDb();

  const result = await db.query(`
    SELECT channel_id AS channelId,
      channel_title AS channelTitle,
      COUNT(*) AS videoCount,
      MAX(created_at) AS importedAt
    FROM videos
    WHERE youtube_id <> ''
      AND file_path = ''
      AND channel_id <> ''
    GROUP BY channel_id, channel_title
    ORDER BY channel_title COLLATE NOCASE
  `);

  return result.values || [];
}

async function androidToggleBookmark(id) {
  const db = await getAndroidDb();

  const existing = await db.query(
    'SELECT id FROM bookmarks WHERE video_id = ?',
    [Number(id)]
  );

  if (existing.values?.length) {
    await db.run('DELETE FROM bookmarks WHERE video_id = ?', [Number(id)]);
  } else {
    await db.run(
      'INSERT OR IGNORE INTO bookmarks(video_id) VALUES(?)',
      [Number(id)]
    );
  }

  return androidListVideos();
}

async function androidReadNotes(notePath) {

  if (!match) return '';

async function androidRecordProgress({
  videoId,
  positionSeconds = 0,
  durationSeconds = 0
}) {
  const db = await getAndroidDb();
  const position = Math.max(0, Number(positionSeconds) || 0);
  const duration = Math.max(0, Number(durationSeconds) || 0);
  const completed = duration > 0 && position / duration >= 0.9 ? 1 : 0;

  await db.run(`
    INSERT INTO watch_history (
      video_id, position_seconds, duration_seconds,
      completed, last_watched_at
    )
    VALUES (?, ?, ?, ?, CURRENT_TIMESTAMP)
    ON CONFLICT(video_id)
    DO UPDATE SET
      position_seconds = excluded.position_seconds,
      duration_seconds = excluded.duration_seconds,
      completed = excluded.completed,
      last_watched_at = CURRENT_TIMESTAMP
  `, [Number(videoId), position, duration, completed]);

  return { success: true, completed: Boolean(completed) };
}

async function androidGetHistory(limit = 30) {
  const db = await getAndroidDb();

  const result = await db.query(`
    SELECT v.*,
      COALESCE(h.position_seconds, 0) AS position_seconds,
      COALESCE(h.duration_seconds, 0) AS duration_seconds,
      COALESCE(h.completed, 0) AS completed,
      h.last_watched_at,
      EXISTS (
        SELECT 1 FROM bookmarks b WHERE b.video_id = v.id
      ) AS bookmarked
    FROM watch_history h
    INNER JOIN videos v ON v.id = h.video_id
    ORDER BY h.last_watched_at DESC
    LIMIT ?
  `, [Math.max(1, Math.min(100, Number(limit) || 30))]);

  return (result.values || []).map(mapVideo);
}

async function androidGetRecommendations(limit = 20) {
  const db = await getAndroidDb();

  const result = await db.query(`
    SELECT v.*,
      EXISTS (
        SELECT 1 FROM bookmarks b WHERE b.video_id = v.id
      ) AS bookmarked,
      COALESCE(h.position_seconds, 0) AS position_seconds,
      COALESCE(h.duration_seconds, 0) AS duration_seconds,
      COALESCE(h.completed, 0) AS completed
    FROM videos v
    LEFT JOIN watch_history h ON h.video_id = v.id
    WHERE v.id NOT IN (
      SELECT video_id FROM watch_history
      WHERE completed = 1
    )
    ORDER BY
      CASE WHEN EXISTS (
        SELECT 1 FROM bookmarks b WHERE b.video_id = v.id
      ) THEN 0 ELSE 1 END,
      CASE WHEN h.video_id IS NULL THEN 0 ELSE 1 END,
      v.created_at DESC
    LIMIT ?
  `, [Math.max(1, Math.min(100, Number(limit) || 20))]);

  return (result.values || []).map(mapVideo);
}

async function androidImportFiles() {
  const db = await getAndroidDb();
  const pickerResult = await FilePicker.pickVideos({ limit: 0 });
  const files = pickerResult?.files || [];

  if (!files.length) return { addedCount: 0, errors: [] };

  let addedCount = 0;
  const errors = [];

  for (const file of files) {
    try {
      if (!file.path) {
        throw new Error(`No accessible URI for ${file.name || 'video'}`);
      }

      const originalName = file.name || 'video';
      const title = originalName.replace(/\.[^/.]+$/, '') || 'Untitled video';

      // Skip a source URI already imported into this library.
      const duplicate = await db.query(
        'SELECT id FROM videos WHERE source_uri = ? LIMIT 1',
        [String(file.path)]
      );

      if (duplicate.values?.length) {
        continue;
      }

      const copied = await LocalMedia.copyVideo({
        uri: file.path,
        fileName: originalName
      });

      if (!copied?.path) {
        throw new Error('Android did not return the copied video path.');
      }

      await db.run(`
        INSERT INTO videos (
          title, channel_title, channel_id, youtube_id,
          youtube_url, file_path, thumbnail_url, description, tags,
          source_uri
        ) VALUES (?, '', '', '', '', ?, '', '', '', ?)
      `, [title, copied.path, String(file.path)]);

      addedCount++;
    } catch (error) {
      console.error('[LocalTube] Import failed:', file.name, error);
      errors.push(`${file.name || 'Video'}: ${error.message || error}`);
    }
  }

  return { addedCount, errors };
}


// Android YouTube Data API v3 channel importer.
async function androidFetchChannel({ channelUrl, apiKey }) {
  const key = String(apiKey || '').trim();
  const input = String(channelUrl || '').trim();

  if (!key) throw new Error('Enter your YouTube Data API key.');
  if (!input) throw new Error('Enter a YouTube channel URL or handle.');

  let channelId = '';
  let handle = '';

  const idMatch = input.match(/UC[a-zA-Z0-9_-]{22}/);

  if (idMatch) {
    channelId = idMatch[0];
  } else {
    const handleMatch = input.match(/@([a-zA-Z0-9._-]+)/);

    if (handleMatch) {
      handle = handleMatch[1];
    } else if (/^@?[a-zA-Z0-9._-]+$/.test(input)) {
      handle = input.replace(/^@/, '');
    } else {
      throw new Error(
        'Enter a valid YouTube channel URL, UC channel ID, or @handle.'
      );
    }
  }

  const base = 'https://www.googleapis.com/youtube/v3/';

  async function youtubeRequest(endpoint, params) {
    const url = new URL(base + endpoint);

    Object.entries({ ...params, key }).forEach(([name, value]) => {
      if (value !== undefined && value !== null && value !== '') {
        url.searchParams.set(name, String(value));
      }
    });

    let response;
    let data;

    try {
      response = await fetch(url.toString());
      data = await response.json();
    } catch (error) {
      throw new Error(
        'Could not reach YouTube. Check your internet connection and try again.'
      );
    }

    if (!response.ok || data.error) {
      const reason = data.error?.errors?.[0]?.reason;
      const message = data.error?.message || 'YouTube API request failed.';
      throw new Error(reason ? `${message} (${reason})` : message);
    }

    return data;
  }

  const channelData = await youtubeRequest('channels', {
    part: 'id,snippet,contentDetails',
    ...(channelId ? { id: channelId } : { forHandle: handle })
  });

  const channel = channelData.items?.[0];

  if (!channel) {
    throw new Error('Channel not found. Check the URL, handle and API key.');
  }

  const uploadsPlaylist =
    channel.contentDetails?.relatedPlaylists?.uploads;

  if (!uploadsPlaylist) {
    throw new Error('Could not find this channel uploads playlist.');
  }

  const collected = [];
  const seenIds = new Set();
  let pageToken = '';

  // Five pages x 50 items = up to 250 recent uploads.
  for (let page = 0; page < 5; page++) {
    const pageData = await youtubeRequest('playlistItems', {
      part: 'snippet,contentDetails',
      playlistId: uploadsPlaylist,
      maxResults: 50,
      ...(pageToken ? { pageToken } : {})
    });

    for (const entry of pageData.items || []) {
      const snippet = entry.snippet || {};
      const youtubeId =
        entry.contentDetails?.videoId ||
        snippet.resourceId?.videoId;

      if (!youtubeId || seenIds.has(youtubeId)) continue;
      seenIds.add(youtubeId);

      collected.push({
        title: snippet.title || 'Untitled',
        channelTitle: channel.snippet?.title || '',
        channelId: channel.id,
        youtubeId,
        youtubeUrl: `https://www.youtube.com/watch?v=${youtubeId}`,
        thumbnailUrl:
          snippet.thumbnails?.medium?.url ||
          snippet.thumbnails?.high?.url ||
          snippet.thumbnails?.default?.url ||
          '',
        description: snippet.description || ''
      });
    }

    pageToken = pageData.nextPageToken || '';
    if (!pageToken) break;
  }

  const db = await getAndroidDb();
  let addedCount = 0;
  let skippedCount = 0;

  for (const item of collected) {
    const existing = await db.query(
      'SELECT id FROM videos WHERE youtube_id = ? LIMIT 1',
      [item.youtubeId]
    );

    if (existing.values?.length) {
      skippedCount++;
      continue;
    }

    await db.run(`
      INSERT OR IGNORE INTO videos (
        title, channel_title, channel_id,
        youtube_id, youtube_url, file_path,
        thumbnail_url, description, tags
      )
      VALUES (?, ?, ?, ?, ?, '', ?, ?, '')
    `, [
      item.title,
      item.channelTitle,
      item.channelId,
      item.youtubeId,
      item.youtubeUrl,
      item.thumbnailUrl,
      item.description
    ]);

    // The earlier SELECT established that this ID was not present.
    // Count it once after attempting the insert.
    addedCount++;
  }

  return {
    channelTitle: channel.snippet?.title || '',
    channelId: channel.id,
    addedCount,
    skippedCount,
    totalFetched: collected.length,
    videos: await androidListVideos({
      type: 'youtube',
      channelId: channel.id
    })
  };
}async function androidOpenPlayer(videoId) {
  const db = await getAndroidDb();

  const result = await db.query(
    `SELECT id, title, file_path, youtube_id
     FROM videos WHERE id = ? LIMIT 1`,
    [Number(videoId)]
  );

  const video = result.values?.[0];

  if (!video) {
    throw new Error('Video not found in your LocalTube library.');
  }

  if (!video.file_path) {
    if (video.youtube_id) {
      return YouTubePlayer.play({
        videoId: video.youtube_id,
        title: video.title || 'LocalTube'
      });
    }
    throw new Error('This video does not have a local file.');
  }

  const id = String(video.id);

  // Recover a checkpoint left by an earlier player session or app interruption.
  let checkpoint = { positionMs: 0, durationMs: 0 };

  try {
    checkpoint = await LocalPlayer.getCheckpoint({ videoId: id });
  } catch (error) {
    console.warn('Could not recover native playback checkpoint:', error);
  }

  const resumePositionMs = Math.max(
    0,
    Number(checkpoint?.positionMs) || 0
  );

  const checkpointDurationSeconds =
    Math.max(0, Number(checkpoint?.durationMs) || 0) / 1000;

  // Synchronize recovered native progress into SQLite.
  if (resumePositionMs > 0 || checkpointDurationSeconds > 0) {
    await androidRecordProgress({
      videoId: video.id,
      positionSeconds: resumePositionMs / 1000,
      durationSeconds: checkpointDurationSeconds
    });
  }

  const playbackResult = await LocalPlayer.play({
    path: video.file_path,
    title: video.title || 'LocalTube',
    videoId: id,
    resumePositionMs
  });

  // Save the final position returned by the native player.
  if (playbackResult && playbackResult.videoId) {
    await androidRecordProgress({
      videoId: video.id,
      positionSeconds: playbackResult.positionSeconds,
      durationSeconds: playbackResult.durationSeconds
    });
  }

  try {
    await LocalPlayer.clearCheckpoint({ videoId: id });
  } catch (error) {
    console.warn('Could not clear native playback checkpoint:', error);
  }

  return playbackResult;
}

export function createPlatformApi() {
  // Electron must take priority because Capacitor reports "web"
  // inside the Electron renderer.
  if (window.localTube) return window.localTube;

  if (Capacitor.getPlatform() === 'android') {
    return createAndroidApi();
  }

  return {
    listVideos: unavailable('Browser database preview'),
    listChannels: unavailable('Browser database preview'),
    toggleBookmark: unavailable('Browser database preview'),
    updateTags: unavailable('Browser database preview'),
    loadNotesPath: unavailable('Browser database preview'),

  };
}

        bookmarkedOnly: true

function VideoGrid({ videos, selected, onSelect, onPlay }) {
  if (!videos.length) {
    return (
      <div className="empty-state">
        <p>No videos found in this section.</p>
      </div>
    );
  }

  return (
    <div className="video-grid">
      {videos.map(video => (
        <button
          key={video.id}
          type="button"
          className={`video-card ${
            selected?.id === video.id ? 'selected' : ''
          }`}
          onClick={() => {
            onSelect(video);
            onPlay(video);
          }}
          title={`Play ${video.title}`}
        >
          <div className="thumb">
            {video.thumbnail_url ? (
              <img
                src={video.thumbnail_url}
                alt=""
                loading="lazy"
              />
            ) : (
              <div className="thumb-placeholder">
                <Play size={27} />
              </div>
            )}

            <span
              className={`source-chip ${
                video.isLocal ? 'local' : 'online'
              }`}
            >
              {video.isLocal ? 'LOCAL FILE' : 'YOUTUBE'}
            </span>

            <span className="card-play-icon">
              <Play size={25} fill="currentColor" />
            </span>
          </div>

          <div className="video-card-copy">
            <strong>{video.title}</strong>
            <span>{video.channel_title || 'Local videos'}</span>
          </div>
        </button>
      ))}
    </div>
  );
}

function getDb() {
  if (db) return db;

  
  fs.mkdirSync(dataDir, { recursive: true });

  db = new Database(path.join(dataDir, 'localtube.sqlite'));
  db.pragma('journal_mode = WAL');
  db.pragma('foreign_keys = ON');

  db.exec(`
    CREATE TABLE IF NOT EXISTS videos (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      title TEXT NOT NULL,
      channel_title TEXT DEFAULT '',
      channel_id TEXT DEFAULT '',
      youtube_id TEXT DEFAULT '',
      youtube_url TEXT DEFAULT '',
      file_path TEXT DEFAULT '',
      thumbnail_url TEXT DEFAULT '',
      description TEXT DEFAULT '',
      tags TEXT DEFAULT '',
      created_at TEXT DEFAULT CURRENT_TIMESTAMP
    );

    CREATE UNIQUE INDEX IF NOT EXISTS videos_unique_local_path
    ON videos(file_path) WHERE file_path <> '';

    CREATE TABLE IF NOT EXISTS bookmarks (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      video_id INTEGER NOT NULL UNIQUE,
      created_at TEXT DEFAULT CURRENT_TIMESTAMP,
      FOREIGN KEY(video_id) REFERENCES videos(id) ON DELETE CASCADE
    );

    CREATE TABLE IF NOT EXISTS settings (
      key TEXT PRIMARY KEY,
      value TEXT NOT NULL
    );

    CREATE TABLE IF NOT EXISTS watch_history (
      video_id INTEGER PRIMARY KEY,
      position_seconds REAL NOT NULL DEFAULT 0,
      duration_seconds REAL NOT NULL DEFAULT 0,
      completed INTEGER NOT NULL DEFAULT 0,
      last_watched_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
      FOREIGN KEY(video_id) REFERENCES videos(id) ON DELETE CASCADE
    );

    CREATE INDEX IF NOT EXISTS watch_history_recent
    ON watch_history(last_watched_at DESC);

    CREATE VIRTUAL TABLE IF NOT EXISTS video_search
    USING fts5(title, channel_title, description, tags,
               content='videos', content_rowid='id');

    CREATE TRIGGER IF NOT EXISTS videos_ai
    AFTER INSERT ON videos BEGIN
      INSERT INTO video_search(rowid,title,channel_title,description,tags)
      VALUES(new.id,new.title,new.channel_title,new.description,new.tags);
    END;

    CREATE TRIGGER IF NOT EXISTS videos_ad
    AFTER DELETE ON videos BEGIN
      INSERT INTO video_search(video_search,rowid,title,channel_title,description,tags)
      VALUES('delete',old.id,old.title,old.channel_title,old.description,old.tags);
    END;

    CREATE TRIGGER IF NOT EXISTS videos_au
    AFTER UPDATE ON videos BEGIN
      INSERT INTO video_search(video_search,rowid,title,channel_title,description,tags)
      VALUES('delete',old.id,old.title,old.channel_title,old.description,old.tags);

      INSERT INTO video_search(rowid,title,channel_title,description,tags)
      VALUES(new.id,new.title,new.channel_title,new.description,new.tags);
    END;
  `);

  return db;
}

function mapVideo(row) {
  return {
    ...row,
    isLocal: Boolean(row.file_path),
    bookmarked: Boolean(row.bookmarked),
    fileExists: Boolean(row.file_path && fs.existsSync(row.file_path))
  };
}

// Accepts either an options object or the older (query, bookmarkedOnly) arguments.
function listVideos(options = {}, oldBookmarkedOnly = false) {
  if (typeof options === 'string') {
    options = {
      query: options,
      bookmarkedOnly: oldBookmarkedOnly
    };
  }

  const {
    type = 'all',
    channelId = '',
    query = '',
    bookmarkedOnly = false
  } = options || {};

  const database = getDb();

  let sql = `
    SELECT v.*,
      EXISTS(
        SELECT 1 FROM bookmarks b WHERE b.video_id = v.id
      ) AS bookmarked
    FROM videos v
    WHERE 1 = 1
  `;

  const params = [];

  if (type === 'local') {
    sql += ` AND v.file_path <> ''`;
  } else if (type === 'youtube') {
    sql += ` AND v.youtube_id <> '' AND v.file_path = ''`;
  }

  if (channelId) {
    sql += ` AND v.channel_id = ?`;
    params.push(String(channelId));
  }

  if (bookmarkedOnly) {
    sql += ` AND EXISTS (
      SELECT 1 FROM bookmarks b WHERE b.video_id = v.id
    )`;
  }

  if (query && String(query).trim()) {
    sql += ` AND (
      v.title LIKE ? OR
      v.channel_title LIKE ? OR
      v.tags LIKE ? OR
      v.description LIKE ?
    )`;

    const search = `%${String(query).trim()}%`;
    params.push(search, search, search, search);
  }

  sql += ` ORDER BY v.created_at DESC, v.id DESC LIMIT 1000`;

  return database.prepare(sql).all(...params).map(mapVideo);
}

function listChannels() {
  return getDb().prepare(`
    SELECT
      channel_id AS channelId,
      channel_title AS channelTitle,
      COUNT(*) AS videoCount,
      MAX(created_at) AS importedAt
    FROM videos
    WHERE youtube_id <> ''
      AND file_path = ''
      AND channel_id <> ''
    GROUP BY channel_id, channel_title
    ORDER BY channel_title COLLATE NOCASE
  `).all();
}

// Save playback progress reported by the player window.
ipcMain.on('player:progress', (_event, payload = {}) => {
  try {
    const videoId = Number(payload.videoId);
    const position = Math.max(0, Number(payload.positionSeconds) || 0);
    const duration = Math.max(0, Number(payload.durationSeconds) || 0);

    if (!Number.isInteger(videoId) || videoId <= 0) return;

    const exists = getDb().prepare(
      'SELECT id FROM videos WHERE id = ?'
    ).get(videoId);

    if (!exists) return;

    const completed = duration > 0 && position / duration >= 0.9 ? 1 : 0;

    getDb().prepare(`
      INSERT INTO watch_history(
        video_id, position_seconds, duration_seconds,
        completed, last_watched_at
      )
      VALUES(?, ?, ?, ?, CURRENT_TIMESTAMP)
      ON CONFLICT(video_id) DO UPDATE SET
        position_seconds = excluded.position_seconds,
        duration_seconds = excluded.duration_seconds,
        completed = excluded.completed,
        last_watched_at = CURRENT_TIMESTAMP
    `).run(videoId, position, duration, completed);

    if (mainWindow && !mainWindow.isDestroyed()) {
      mainWindow.webContents.send('history:updated');
    }
  } catch (error) {
    console.error('[LocalTube] Could not save playback progress:', error);
  }
});

      const filePath = path.resolve(selectedPath);

        filePath,
        path.extname(filePath)
      ).replace(/[._-]+/g, ' ');

      insert.run(title, filePath);
      added++;
    }

    return added;
  });

  const addedCount = transaction(result.filePaths);

  return {
    addedCount,
    videos: listVideos({ type: 'local' })

// Import a YouTube channel using YouTube Data API v3.
ipcMain.handle('channels:fetch', async (_event, payload = {}) => {
  const channelUrl = payload.channelUrl || '';
  const apiKey = payload.apiKey || '';

  if (!String(apiKey).trim()) {
    throw new Error('Enter your YouTube Data API v3 key first.');
  }

  const input = String(channelUrl).trim();

  if (!input) {
    throw new Error('Enter a YouTube channel URL or @handle.');
  }

  let channelId = '';
  let handle = '';

  const channelUrlMatch = input.match(
    /youtube\.com\/channel\/(UC[\w-]+)/i
  );

  const handleUrlMatch = input.match(
    /youtube\.com\/@([\w.-]+)/i
  );

  if (channelUrlMatch) {
    channelId = channelUrlMatch[1];
  } else if (handleUrlMatch) {
    handle = handleUrlMatch[1];
  } else if (/^UC[\w-]+$/.test(input)) {
    channelId = input;
  } else if (/^@?[\w.-]+$/.test(input)) {
    handle = input.replace(/^@/, '');
  } else {
    throw new Error(
      'Invalid channel input. Use a channel URL, @handle, or UC channel ID.'
    );
  }

  const baseUrl = 'https://www.googleapis.com/youtube/v3/';

  async function youtubeRequest(endpoint, params) {
    const url = new URL(endpoint, baseUrl);

    for (const [key, value] of Object.entries({
      ...params,
      key: String(apiKey).trim()
    })) {
      url.searchParams.set(key, String(value));
    }

    const response = await fetch(url);
    const data = await response.json();

    if (!response.ok) {
      throw new Error(
        data.error?.message || `YouTube API error: ${response.status}`
      );
    }

    return data;
  }

  let channelData;

  if (handle) {
    channelData = await youtubeRequest('channels', {
      part: 'id,snippet,contentDetails',
      forHandle: handle
    });
  } else {
    channelData = await youtubeRequest('channels', {
      part: 'id,snippet,contentDetails',
      id: channelId
    });
  }

  const channel = channelData.items?.[0];

  if (!channel) {
    throw new Error(
      'Channel not found. Check the channel URL, handle, and API key.'
    );
  }

  const uploadsPlaylist =
    channel.contentDetails?.relatedPlaylists?.uploads;

  if (!uploadsPlaylist) {
    throw new Error('Could not find this channelâ€™s uploads playlist.');
  }

  const collected = [];
  let pageToken = '';

  // Import up to 250 recent uploads.
  for (let page = 0; page < 5; page++) {
    const pageParams = {
      part: 'snippet,contentDetails',
      playlistId: uploadsPlaylist,
      maxResults: 50
    };

    if (pageToken) {
      pageParams.pageToken = pageToken;
    }

    const pageData = await youtubeRequest(
      'playlistItems',
      pageParams
    );

    for (const entry of pageData.items || []) {
      const snippet = entry.snippet || {};

      const youtubeId =
        entry.contentDetails?.videoId ||
        snippet.resourceId?.videoId;

      if (!youtubeId) continue;

      collected.push({
        title: snippet.title || 'Untitled',
        channelTitle: channel.snippet?.title || '',
        channelId: channel.id,
        youtubeId,
        youtubeUrl: `https://www.youtube.com/watch?v=${youtubeId}`,
        thumbnailUrl:
          snippet.thumbnails?.medium?.url ||
          snippet.thumbnails?.high?.url ||
          snippet.thumbnails?.default?.url ||
          '',
        description: snippet.description || ''
      });
    }

    pageToken = pageData.nextPageToken || '';

    if (!pageToken) break;
  }

  const database = getDb();

  const findExisting = database.prepare(
    'SELECT id FROM videos WHERE youtube_id = ?'
  );

  const insert = database.prepare(`
    INSERT INTO videos(
      title, channel_title, channel_id,
      youtube_id, youtube_url, thumbnail_url,
      description, tags
    )
    VALUES(
      @title, @channelTitle, @channelId,
      @youtubeId, @youtubeUrl, @thumbnailUrl,
      @description, ''
    )
  `);

  const transaction = database.transaction((rows) => {
    let added = 0;

    for (const item of rows) {
      if (findExisting.get(item.youtubeId)) continue;

      insert.run({
        title: item.title,
        channelTitle: item.channelTitle,
        channelId: item.channelId,
        youtubeId: item.youtubeId,
        youtubeUrl: item.youtubeUrl,
        thumbnailUrl: item.thumbnailUrl,
        description: item.description
      });

      added++;
    }

    return added;
  });

  const addedCount = transaction(collected);

  return {
    channelTitle: channel.snippet?.title || '',
    channelId: channel.id,
    addedCount,
    videos: listVideos({
      type: 'youtube',
      channelId: channel.id
    })
  };
});

ipcMain.handle('player:open', async (_event, videoId) => {
  const database = getDb();

  const video = database
    .prepare('SELECT * FROM videos WHERE id = ?')
    .get(Number(videoId));

  if (!video) {
    throw new Error('Video not found in LocalTube library.');
  }

  const savedProgress = database.prepare(`
    SELECT position_seconds, duration_seconds, completed
    FROM watch_history
    WHERE video_id = ?
  `).get(Number(videoId));

  const resumeSeconds =
    savedProgress && !savedProgress.completed
      ? Math.max(0, Number(savedProgress.position_seconds) || 0)
      : 0;

  // Determine the playback source before creating the player window.
  let playerType;
  let playerSource;

  if (video.file_path) {
    if (!fs.existsSync(video.file_path)) {
      throw new Error('Local video file does not exist.');
    }

    playerType = 'local';
    playerSource = pathToFileURL(video.file_path).href;
  } else if (video.youtube_id) {
    playerType = 'youtube';
    playerSource = String(video.youtube_id).trim();

    if (!/^[a-zA-Z0-9_-]{11}$/.test(playerSource)) {
      throw new Error('This catalogue entry has an invalid YouTube video ID.');
    }
  } else {
    throw new Error('This video has no playable source.');
  }

  console.log('[LocalTube] Opening player:', {
    videoId: Number(videoId),
    playerType,
    title: video.title
  });

  const playerWindow = new BrowserWindow({
    width: 1200,
    height: 760,
    minWidth: 800,
    minHeight: 500,
    backgroundColor: '#080a12',
    title: `LocalTube Player - ${video.title || 'Video'}`,
    autoHideMenuBar: true,
    webPreferences: {
      preload: path.join(__dirname, 'player-preload.cjs'),
      contextIsolation: true,
      nodeIntegration: false,
      sandbox: true,
      webSecurity: true,
      allowRunningInsecureContent: false,
      webviewTag: false
    }
  });

  // Do not allow links inside the embedded player to create new windows.
  playerWindow.webContents.setWindowOpenHandler(({ url }) => {
    console.log('[LocalTube Security] Blocked player pop-up:', url);
    return { action: 'deny' };
  });

  // The player page itself must never navigate to another page.
  playerWindow.webContents.on('will-navigate', (event, url) => {
    console.log('[LocalTube Security] Blocked player navigation:', url);
    event.preventDefault();
  });



  // Block redirects of the top-level LocalTube player page.
  playerWindow.webContents.on(
    'will-redirect',
    (event, url, _isInPlace, isMainFrame) => {
      if (isMainFrame) {
        console.log('[LocalTube Security] Blocked player redirect:', url);
        event.preventDefault();
      }
    }
  );

  // Supply a Referer for YouTube's embedded-player request (error 153).
  // Register this only once; repeated player launches must not stack listeners.
  if (playerType === 'youtube' && !globalThis.__localTubeRefererHandlerInstalled) {
    playerWindow.webContents.session.webRequest.onBeforeSendHeaders(
      {
        urls: [
          'https://www.youtube-nocookie.com/*',
          'https://www.youtube.com/*'
        ]
      },
      (details, callback) => {
        details.requestHeaders.Referer = 'http://localhost/';
        callback({ requestHeaders: details.requestHeaders });
      }
    );

    globalThis.__localTubeRefererHandlerInstalled = true;
  }

  playerWindow.on('closed', () => {
    console.log('[LocalTube] Player window closed.');
  });

  try {
    await playerWindow.loadFile(
      path.join(__dirname, 'player.html'),
      {
        query: {
          type: playerType,
          src: playerSource,
          title: video.title || 'LocalTube Player',
          videoId: String(video.id),
          start: String(resumeSeconds)
        }
      }
    );
  } catch (error) {
    if (!playerWindow.isDestroyed()) {
      playerWindow.close();
    }
    throw error;
  }

  return { success: true };
});

function saveProgress(position, duration) {
  if (!videoId || !window.localTubePlayer?.saveProgress) return;

  const now = Date.now();
  if (now - lastSavedAt < 4000) return;
  lastSavedAt = now;

  window.localTubePlayer.saveProgress({
    videoId,
    positionSeconds: Math.max(0, Number(position) || 0),
    durationSeconds: Math.max(0, Number(duration) || 0)
  });
}

function stopProgressTimer() {
  if (progressTimer) {
    clearInterval(progressTimer);
    progressTimer = null;
  }
}

function startProgressTimer(callback) {
  stopProgressTimer();
  progressTimer = setInterval(callback, 3000);
}

if (type === 'local' && src) {
  video.style.display = 'block';
  status.style.display = 'none';
  video.src = src;

  video.addEventListener('loadedmetadata', () => {
    if (startSeconds > 0 && startSeconds < video.duration) {
      video.currentTime = startSeconds;
    }
  });

  video.addEventListener('play', () => {
    startProgressTimer(() => {
      if (!video.paused && !video.ended) {
        saveProgress(video.currentTime, video.duration);
      }
    });
  });

  video.addEventListener('pause', () => {
    stopProgressTimer();
    saveProgress(video.currentTime, video.duration);
  });

  video.addEventListener('timeupdate', () => {
    if (video.ended) {
      saveProgress(video.duration, video.duration);
    }
  });

  video.addEventListener('ended', () => {
    stopProgressTimer();
    saveProgress(video.duration, video.duration);
  });

  window.addEventListener('beforeunload', () => {
    saveProgress(video.currentTime, video.duration);
  });

  video.addEventListener('error', () => {

  });

package com.localtube.app;

import android.net.Uri;
import android.webkit.MimeTypeMap;

import com.getcapacitor.JSObject;
import com.getcapacitor.Plugin;
import com.getcapacitor.PluginCall;
import com.getcapacitor.PluginMethod;
import com.getcapacitor.annotation.CapacitorPlugin;

import java.io.File;
import java.io.FileOutputStream;
import java.io.InputStream;
import java.io.IOException;

@CapacitorPlugin(name = "LocalMedia")
public class LocalMediaPlugin extends Plugin {

    @PluginMethod
    public void copyVideo(PluginCall call) {
        String source = call.getString("uri");
        String requestedName = call.getString("fileName", "video");

        if (source == null || source.trim().isEmpty()) {
            call.reject("A video URI is required.");
            return;
        }

        try {
            Uri sourceUri = Uri.parse(source);
            File videoDirectory = new File(
                getContext().getFilesDir(),
                "LocalTube/videos"
            );

            if (!videoDirectory.exists() && !videoDirectory.mkdirs()) {
                call.reject("Could not create LocalTube video storage.");
                return;
            }

            String safeName = requestedName
                .replaceAll("[^a-zA-Z0-9._-]", "_");

            if (safeName.isEmpty() || safeName.equals(".")) {
                safeName = "video";
            }

            File destination = new File(videoDirectory, safeName);

            String baseName = safeName;
            int dot = safeName.lastIndexOf('.');
            String extension = "";

            if (dot > 0) {
                baseName = safeName.substring(0, dot);
                extension = safeName.substring(dot);
            }

            int suffix = 1;
            while (destination.exists()) {
                destination = new File(
                    videoDirectory,
                    baseName + "_" + suffix + extension
                );
                suffix++;
            }

            long totalBytes = 0;
            byte[] buffer = new byte[64 * 1024];

            try (
                InputStream input = getContext()
                    .getContentResolver()
                    .openInputStream(sourceUri);
                FileOutputStream output = new FileOutputStream(destination)
            ) {
                if (input == null) {
                    call.reject("Android could not open the selected video.");
                    return;
                }

                int bytesRead;
                while ((bytesRead = input.read(buffer)) != -1) {
                    output.write(buffer, 0, bytesRead);
                    totalBytes += bytesRead;
                }

                output.flush();
            } catch (Exception copyError) {
                if (destination.exists()) {
                    destination.delete();
                }
                throw copyError;
            }

            if (totalBytes == 0) {
                destination.delete();
                call.reject("The selected video is empty.");
                return;
            }

            JSObject result = new JSObject();
            result.put("path", destination.getAbsolutePath());
            result.put("fileName", destination.getName());
            result.put("size", totalBytes);

            call.resolve(result);

        } catch (Exception error) {
            call.reject("Video import failed: " + error.getMessage(), error);
        }
    }

    @PluginMethod
    public void fileExists(PluginCall call) {
        String path = call.getString("path");

        boolean exists = false;

        if (path != null && !path.trim().isEmpty()) {
            File file = new File(path);
            exists = file.exists() && file.isFile() && file.canRead();
        }

        JSObject result = new JSObject();
        result.put("exists", exists);
        call.resolve(result);
    }}

package com.localtube.app;

import android.content.Intent;
import android.text.TextUtils;

import androidx.activity.result.ActivityResult;

import com.getcapacitor.JSObject;
import com.getcapacitor.Plugin;
import com.getcapacitor.PluginCall;
import com.getcapacitor.PluginMethod;
import com.getcapacitor.annotation.ActivityCallback;
import com.getcapacitor.annotation.CapacitorPlugin;

import java.io.File;

@CapacitorPlugin(name = "LocalPlayer")
public class LocalPlayerPlugin extends Plugin {

    private static final String PREFS = "localtube_playback";

    @PluginMethod
    public void play(PluginCall call) {
        String path = call.getString("path");
        String title = call.getString("title", "LocalTube");
        String videoId = call.getString("videoId");
        long resumePositionMs = call.getLong("resumePositionMs", 0L);

        if (TextUtils.isEmpty(path)) {
            call.reject("Video path is missing.");
            return;
        }

        if (TextUtils.isEmpty(videoId)) {
            call.reject("Video ID is missing.");
            return;
        }

        File videoFile = new File(path);

        if (!videoFile.exists() || !videoFile.isFile()) {
            call.reject("The imported video file could not be found.");
            return;
        }

        try {
            Intent intent = new Intent(getActivity(), LocalPlayerActivity.class);

            intent.putExtra("videoPath", videoFile.getAbsolutePath());
            intent.putExtra("videoTitle", title);
            intent.putExtra("videoId", videoId);
            intent.putExtra("resumePositionMs", Math.max(0L, resumePositionMs));

            startActivityForResult(call, intent, "handlePlayerResult");
        } catch (Exception error) {
            call.reject("Could not open the native video player: " + error.getMessage(), error);
        }
    }

    @PluginMethod
    public void getCheckpoint(PluginCall call) {
        String videoId = call.getString("videoId");

        if (TextUtils.isEmpty(videoId)) {
            call.reject("Video ID is missing.");
            return;
        }

        android.content.SharedPreferences prefs =
            getContext().getSharedPreferences(PREFS, android.content.Context.MODE_PRIVATE);

        long position = prefs.getLong("position_" + videoId, 0L);
        long duration = prefs.getLong("duration_" + videoId, 0L);

        if (duration > 0L && position >= duration * 0.9) {
            position = 0L;
        }

        JSObject response = new JSObject();
        response.put("positionMs", position);
        response.put("durationMs", duration);
        call.resolve(response);
    }

    @PluginMethod
    public void clearCheckpoint(PluginCall call) {
        String videoId = call.getString("videoId");

        if (TextUtils.isEmpty(videoId)) {
            call.reject("Video ID is missing.");
            return;
        }

        android.content.SharedPreferences prefs =
            getContext().getSharedPreferences(PREFS, android.content.Context.MODE_PRIVATE);

        prefs.edit()
            .remove("position_" + videoId)
            .remove("duration_" + videoId)
            .apply();

        call.resolve();
    }

    @ActivityCallback
    private void handlePlayerResult(PluginCall call, ActivityResult result) {
        if (call == null) return;

        Intent data = result.getData();

        if (data == null) {
            call.resolve();
            return;
        }

        String videoId = data.getStringExtra("videoId");
        long positionMs = Math.max(0L, data.getLongExtra("positionMs", 0L));
        long durationMs = Math.max(0L, data.getLongExtra("durationMs", 0L));

        JSObject response = new JSObject();
        response.put("videoId", videoId);
        response.put("positionSeconds", positionMs / 1000.0);
        response.put("durationSeconds", durationMs / 1000.0);
        response.put("completed", durationMs > 0L && positionMs >= durationMs * 0.9);

        call.resolve(response);
    }
}

package com.localtube.app;

import android.content.Context;
import android.content.SharedPreferences;
import android.content.Intent;
import android.net.Uri;
import android.os.Bundle;
import android.os.Handler;
import android.os.Looper;
import android.view.ViewGroup;
import android.widget.FrameLayout;
import android.graphics.Color;

import androidx.appcompat.app.AppCompatActivity;
import androidx.media3.common.MediaItem;
import androidx.media3.exoplayer.ExoPlayer;
import androidx.media3.ui.PlayerView;

import java.io.File;

public class LocalPlayerActivity extends AppCompatActivity {

    private static final String PREFS = "localtube_playback";

    private ExoPlayer player;
    private PlayerView playerView;

    private final Handler progressHandler = new Handler(Looper.getMainLooper());

    private String videoId;
    private long lastPositionMs = 0L;
    private long lastDurationMs = 0L;
    private boolean resultSent = false;

    private final Runnable checkpointRunnable = new Runnable() {
        @Override
        public void run() {
            saveCheckpoint();
            progressHandler.postDelayed(this, 5000L);
        }
    };

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);

        String path = getIntent().getStringExtra("videoPath");
        String title = getIntent().getStringExtra("videoTitle");
        videoId = getIntent().getStringExtra("videoId");

        long resumePositionMs =
            getIntent().getLongExtra("resumePositionMs", 0L);

        if (path == null || path.trim().isEmpty() || videoId == null) {
            setResult(RESULT_CANCELED);
            finish();
            return;
        }

        File videoFile = new File(path);

        if (!videoFile.exists() || !videoFile.isFile()) {
            setResult(RESULT_CANCELED);
            finish();
            return;
        }

        getWindow().setStatusBarColor(Color.BLACK);
        getWindow().setNavigationBarColor(Color.BLACK);

        FrameLayout root = new FrameLayout(this);
        root.setBackgroundColor(Color.BLACK);

        playerView = new PlayerView(this);
        playerView.setUseController(true);
        playerView.setControllerAutoShow(true);
        playerView.setShowBuffering(PlayerView.SHOW_BUFFERING_WHEN_PLAYING);

        FrameLayout.LayoutParams params = new FrameLayout.LayoutParams(
            ViewGroup.LayoutParams.MATCH_PARENT,
            ViewGroup.LayoutParams.MATCH_PARENT
        );

        root.addView(playerView, params);
        setContentView(root);

        setTitle(title == null ? "LocalTube" : title);

        player = new ExoPlayer.Builder(this).build();
        playerView.setPlayer(player);

        player.setMediaItem(MediaItem.fromUri(Uri.fromFile(videoFile)));
        player.prepare();

        if (resumePositionMs > 0L) {
            player.seekTo(resumePositionMs);
        }

        player.play();

        progressHandler.postDelayed(checkpointRunnable, 5000L);
    }

    private void saveCheckpoint() {
        if (player == null || videoId == null) return;

        long position = Math.max(0L, player.getCurrentPosition());
        long duration = Math.max(0L, player.getDuration());

        // Keep the actual playback position for the result returned to SQLite.
        // Only reset the stored resume checkpoint after completion.
        lastPositionMs = position;
        lastDurationMs = duration;

        if (duration > 0L && position >= duration * 0.9) {
            position = 0L;
        }

        SharedPreferences prefs =
            getSharedPreferences(PREFS, Context.MODE_PRIVATE);

        prefs.edit()
            .putLong("position_" + videoId, position)
            .putLong("duration_" + videoId, duration)
            .apply();
    }

    private void returnPlaybackResult() {
        if (resultSent) return;

        saveCheckpoint();
        resultSent = true;

        Intent result = new Intent();
        result.putExtra("videoId", videoId);
        result.putExtra("positionMs", lastPositionMs);
        result.putExtra("durationMs", lastDurationMs);

        setResult(RESULT_OK, result);
    }

    @Override
    protected void onPause() {
        saveCheckpoint();

        if (player != null) {
            player.pause();
        }

        super.onPause();
    }

    @Override
    public void finish() {
        returnPlaybackResult();
        super.finish();
    }

    @Override
    protected void onDestroy() {
        progressHandler.removeCallbacks(checkpointRunnable);
        returnPlaybackResult();

        if (playerView != null) {
            playerView.setPlayer(null);
        }

        if (player != null) {
            player.release();
            player = null;
        }

        super.onDestroy();
    }
}
